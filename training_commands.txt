  LoRA+ 训练命令速查
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

【使用前先激活环境】
conda activate diet-training

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

① 从头训练（跑完所有 epoch）
───────────────────────────────────────────────────────
python train_yolo.py --strategy lora_plus --data datasets/food_cls --model yolov8n-cls.pt --epochs 25 --batch 32 --lr0 0.01 --lora-rank 8 --lora-alpha 16 --lora-plus-ratio 16 --name food_cls_lora

  --epochs 100   ← 想多跑几轮就改这个数字
  --name food_cls_lora  ← 改成你想要的实验名

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

② 训练中断后接着跑（断点续训）
───────────────────────────────────────────────────────
python train_yolo.py --strategy lora_plus --resume --name food_cls_lora

  会自动从上次保存的位置继续，epoch 总数和原来一样。

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

③ 训练完成后还想继续加训
───────────────────────────────────────────────────────
python train_yolo.py --strategy lora_plus --resume --epochs 150 --name food_cls_lora

  把 --epochs 设得比之前大（原来 100，这里写 150 就再加 50 轮），
  会自动从第 101 轮继续跑到 150 轮。

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

④ 生成 Excel 训练报表
───────────────────────────────────────────────────────
python generate_report.py runs/classify/food_cls_lora

  在 runs/classify/food_cls_lora/ 下生成 training_report.xlsx，
  包含汇总（最佳准确率等）和每轮详情两个 Sheet。
